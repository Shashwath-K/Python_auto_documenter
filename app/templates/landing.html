def calculate_lexical_diversity(text: str, remove_stopwords: bool = True) -> dict:
    """
    Computes lexical diversity (Type-Token Ratio) and extracts domain keywords using NLTK.

    Args:
        text (str): Source text or docstrings to analyze.
        remove_stopwords (bool): Whether to filter high-frequency noise words.

    Returns:
        dict: Analytics payload containing token counts, TTR score, and lemma frequencies.
    """
    # NLTK Tokenization & Stopword filtration
    tokens = [t.lower() for t in tokenize_words(text)]
    if remove_stopwords:
        tokens = filter_stopwords(tokens)
    
    unique_tokens = set(tokens)
    ttr = round(len(unique_tokens) / max(len(tokens), 1), 3)
    return {"ttr": ttr, "tokens": tokens, "vocab_size": len(unique_tokens)}
